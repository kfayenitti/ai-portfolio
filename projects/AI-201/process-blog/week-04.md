// Input 0: scattered points
// Output: polyline rays (units) from each point

int rays_per_unit      = chi("rays_per_unit");      // e.g. 4 to 8
float base_max_len     = chf("base_max_len");       // e.g. 0.25
float base_min_len     = chf("base_min_len");       // e.g. 0.04
float clearance        = chf("clearance");          // e.g. 0.03
float safety_margin    = chf("safety_margin");      // e.g. 0.01
float search_radius    = chf("search_radius");      // e.g. 0.6
float angle_jitter_deg = chf("angle_jitter_deg");   // e.g. 12
int seed               = chi("seed");

float jitter = radians(angle_jitter_deg);

int n = npoints(0);

// optional width attr for later polywire
addattrib(0, "point", "pscale", 0.0);

for (int i = 0; i < n; i++)
{
    vector Pi = point(0, "P", i);

    // Use existing scatter point as branch root
    // (keeps point count lower than duplicating roots)
    setpointattrib(0, "pscale", i, chf("root_pscale"), "set");

    // local random rotation so each unit has unique orientation
    float base_ang = rand(set(i, seed, 101)) * M_TWOPI;

    // nearby candidates that can block rays
    int near[] = nearpoints(0, Pi, search_radius);

    for (int r = 0; r < max(1, rays_per_unit); r++)
    {
        float t = float(r) / float(max(1, rays_per_unit)); // 0..1
        float ang = base_ang + t * M_TWOPI;
        ang += fit01(rand(set(i, r, seed)), -jitter, jitter);

        vector dir = set(cos(ang), sin(ang), 0.0);
        dir = normalize(dir);

        float allowed = base_max_len;

        // find closest blocker along this ray
        foreach (int j; near)
        {
            if (j == i) continue;

            vector Pj = point(0, "P", j);
            vector v = Pj - Pi;

            float fwd = dot(v, dir);      // distance along ray
            if (fwd <= 0.0) continue;     // behind root

            // perpendicular distance from neighbor to ray axis
            vector rej = v - dir * fwd;
            float lat = length(rej);

            // if neighbor is close to axis, it can block this ray
            if (lat < clearance)
            {
                float stop = fwd - safety_margin;
                if (stop < allowed)
                    allowed = stop;
            }
        }

        float final_len = clamp(allowed, base_min_len, base_max_len);
        if (final_len <= 1e-5) continue;

        vector tipP = Pi + dir * final_len;
        int tip = addpoint(0, tipP);
        setpointattrib(0, "pscale", tip, chf("tip_pscale"), "set");

        addprim(0, "polyline", i, tip);
    }
}

### What I Learned

- VEX detail wrangles that build geometry in one pass are powerful, but small ordering mistakes (branch range, tangent frames, welding vs new points) show up immediately in the viewport as “wrong” topology.
- When branches looked like they were converging, the fix was not viewport settings—it was revisiting how directions were constructed (parallel offsets along the stem vs ad-hoc rotations).
- Planning growth as a single scalar over time (`growth` 0→1) is straightforward in theory; implementing it cleanly means deciding *when* each knot appears and how tip behavior is handled (I noted the tip-timing issue for later patches like in `generatedFrost_V02.md`).
